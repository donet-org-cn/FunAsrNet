cmake_minimum_required(VERSION 3.18)
project(funasr_native_cpu_reference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE ON CACHE BOOL "" FORCE)
set(GGML_OPENMP ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/llama-source" llama-build)
add_executable(funasr-native-cpu-reference reference-main.cpp)
target_include_directories(funasr-native-cpu-reference PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/funasr-common")
target_compile_definitions(funasr-native-cpu-reference PRIVATE _USE_MATH_DEFINES NOMINMAX)
target_link_libraries(funasr-native-cpu-reference PRIVATE llama ggml)
